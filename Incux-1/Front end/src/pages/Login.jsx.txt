import {
  useState
} from 'react';

import API from '../api/axios';

import {
  useNavigate,
  Link
} from 'react-router-dom';

import {
  useAuth
} from '../context/AuthContext';

export default function Login() {

  const navigate =
    useNavigate();

  const { login } =
    useAuth();

  const [form, setForm] =
    useState({

      email: '',
      password: ''

    });

  const handleSubmit =
    async (e) => {

    e.preventDefault();

    try {

      const res =
        await API.post(
          '/auth/login',
          form
        );

      login(
        res.data.token,
        res.data.user
      );

      if (
        res.data.user.role ===
        'admin'
      ) {

        navigate('/admin');

      } else {

        navigate('/intern');

      }

    } catch (error) {

      alert(
        error.response?.data?.error
      );

    }
  };

  return (

    <div className="min-h-screen flex justify-center items-center">

      <form
        onSubmit={handleSubmit}
        className="bg-white p-8 rounded shadow w-96"
      >

        <h2 className="text-2xl font-bold mb-6">
          Login
        </h2>

        <input
          type="email"
          placeholder="Email"
          className="w-full border p-2 mb-4"
          onChange={(e) =>
            setForm({
              ...form,
              email: e.target.value
            })
          }
        />

        <input
          type="password"
          placeholder="Password"
          className="w-full border p-2 mb-4"
          onChange={(e) =>
            setForm({
              ...form,
              password: e.target.value
            })
          }
        />

        <button
          className="bg-indigo-600 text-white w-full py-2 rounded"
        >
          Login
        </button>

        <Link
          to="/register"
          className="block mt-4 text-center text-indigo-600"
        >
          Create Account
        </Link>

      </form>

    </div>
  );
}
