import {
  useState
} from 'react';

import API from '../api/axios';

import {
  useNavigate
} from 'react-router-dom';

export default function Register() {

  const navigate =
    useNavigate();

  const [form, setForm] =
    useState({

      name: '',
      email: '',
      phone: '',
      password: ''

    });

  const handleSubmit =
    async (e) => {

    e.preventDefault();

    try {

      await API.post(
        '/auth/register',
        form
      );

      alert(
        'Registration Successful'
      );

      navigate('/');

    } catch (error) {

      alert(
        error.response?.data?.error
      );

    }
  };

  return (

    <div className="min-h-screen flex justify-center items-center">

      <form
        onSubmit={handleSubmit}
        className="bg-white p-8 rounded shadow w-96"
      >

        <h2 className="text-2xl font-bold mb-6">
          Register
        </h2>

        <input
          type="text"
          placeholder="Name"
          className="w-full border p-2 mb-4"
          onChange={(e) =>
            setForm({
              ...form,
              name: e.target.value
            })
          }
        />

        <input
          type="email"
          placeholder="Email"
          className="w-full border p-2 mb-4"
          onChange={(e) =>
            setForm({
              ...form,
              email: e.target.value
            })
          }
        />

        <input
          type="text"
          placeholder="Phone"
          className="w-full border p-2 mb-4"
          onChange={(e) =>
            setForm({
              ...form,
              phone: e.target.value
            })
          }
        />

        <input
          type="password"
          placeholder="Password"
          className="w-full border p-2 mb-4"
          onChange={(e) =>
            setForm({
              ...form,
              password: e.target.value
            })
          }
        />

        <button
          className="bg-indigo-600 text-white w-full py-2 rounded"
        >
          Register
        </button>

      </form>

    </div>
  );
}
