const mongoose = require('mongoose');

const AttendanceSchema = new mongoose.Schema({

  userId: {
    type: mongoose.Schema.Types.ObjectId,
    ref: 'User',
    required: true
  },

  date: {
    type: String,
    required: true
  },

  loginTime: {
    type: Date,
    required: true
  },

  logoutTime: {
    type: Date,
    default: null
  },

  lateMinutes: {
    type: Number,
    default: 0
  },

  status: {
    type: String,
    default: 'On Time'
  }

}, {
  timestamps: true
});

AttendanceSchema.index(
  { userId: 1, date: 1 },
  { unique: true }
);

module.exports = mongoose.model(
  'Attendance',
  AttendanceSchema
);
