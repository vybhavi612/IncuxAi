const mongoose = require('mongoose');

const TaskSchema = new mongoose.Schema({

  userId: {
    type: mongoose.Schema.Types.ObjectId,
    ref: 'User',
    required: true
  },

  title: {
    type: String,
    required: true
  },

  category: {
    type: String,
    required: true
  },

  status: {
    type: String,
    enum: ['Pending', 'Completed'],
    default: 'Pending'
  }

}, {
  timestamps: true
});

module.exports = mongoose.model(
  'Task',
  TaskSchema
);
